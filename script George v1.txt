local rotatingBorders = {}
local function AddRotatingBorder(target, thickness, speed)
    if target:FindFirstChildOfClass("UIStroke") then return end
    thickness = thickness or 1.5
    speed     = speed or 90

    local stroke = Instance.new("UIStroke")
    stroke.Thickness       = thickness
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Color           = Color3.new(1, 1, 1)
    stroke.Parent          = target

    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0,    Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.45, Color3.fromRGB(160, 160, 160)),
        ColorSequenceKeypoint.new(1,    Color3.fromRGB(30,  30,  30)),
    })
    grad.Parent = stroke

    table.insert(rotatingBorders, {grad = grad, speed = speed})
    return stroke
end

RunService.Heartbeat:Connect(function(dt)
    for _, b in ipairs(rotatingBorders) do
        if b.grad and b.grad.Parent then
            b.grad.Rotation = (b.grad.Rotation + b.speed * dt) % 360
        end
    end
end)

local function MakeButton(text)
    local btn = Instance.new("TextButton")
    btn.Text                   = text
    btn.Font                   = Enum.Font.SourceSansBold
    btn.TextSize               = 18
    btn.TextColor3             = Color3.fromRGB(255, 255, 255)
    btn.BackgroundColor3       = Color3.fromRGB(18, 18, 18)
    btn.BackgroundTransparency = 0.15
    btn.AutoButtonColor        = false
    btn.Size                   = UDim2.fromOffset(190, 52)
    btn.TextTransparency       = 1
    btn.ZIndex                 = 6
    btn.Parent                 = ButtonsHolder
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 12)

    AddRotatingBorder(btn, 1.4, 100)

    btn.MouseButton1Down:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12, Enum.EasingStyle.Quad), {
            BackgroundColor3       = Color3.fromRGB(40, 40, 40),
            BackgroundTransparency = 0,
        }):Play()
    end)
    btn.MouseButton1Up:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.18, Enum.EasingStyle.Quad), {
            BackgroundColor3       = Color3.fromRGB(18, 18, 18),
            BackgroundTransparency = 0.15,
        }):Play()
    end)
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), { BackgroundTransparency = 0.05 }):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(18, 18, 18),
            BackgroundTransparency = 0.15
        }):Play()
    end)
    return btn
end

local function AntiSpy()
    local player = Players.LocalPlayer
    local PlayerGui = player:WaitForChild("PlayerGui")

    local blacklistedGUIs = {
        ["SimpleSpy2"] = true, ["SimpleSpy1"] = true, ["SimpleSpy"] = true,
        ["Spy"] = true, ["Simple_Spy"] = true, ["Simple-Spy"] = true,
        ["RemoteSpy"] = true, ["Remote_Spy"] = true, ["Remote-Spy"] = true,
        ["RemotesSpy"] = true, ["Remotes_Spy"] = true, ["Remotes-Spy"] = true,
        ["Remotes"] = true, ["Remote"] = true,
    }
    for i = 3, 20 do
        blacklistedGUIs["SimpleSpy"..i] = true
        blacklistedGUIs["Spy"..i]       = true
    end

    local function checkGui(child)
        if blacklistedGUIs[child.Name] then
            Players.LocalPlayer:Kick("Anti-Spy: Removed spy GUI")
        end
    end

    for _, gui in pairs(CoreGui:GetChildren())                    do checkGui(gui) end
    for _, gui in pairs(PlayerGui:GetChildren())                  do checkGui(gui) end
    CoreGui.ChildAdded:Connect(checkGui)
    PlayerGui.ChildAdded:Connect(checkGui)
end
